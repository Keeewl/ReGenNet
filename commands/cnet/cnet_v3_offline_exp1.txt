==================== Train ====================

export CUDA_VISIBLE_DEVICES=4,5,6,7
mpiexec -n 4 --allow-run-as-root python -m train.train_mdm \
  --setting cnet_v3 \
  --save_dir save/cnet_v3/chi3d_smplx_offline_exp1 \
  --dataset chi3d \
  --cond_mask_prob 0 \
  --num_person 2 \
  --layers 8 \
  --num_frames 150 \
  --arch offline \
  --cm_mode concat \
  --overwrite \
  --pose_rep rot6d \
  --body_model smplx \
  --data_path dataset/chi3d/chi3d_smplx_train.h5 \
  --train_platform_type TensorboardPlatform \
  --vel_threshold 0.01 \
  --unconstrained \
  --resume_checkpoint save/cnet_v3/chi3d_smplx_offline_exp1/model000050000.pt
