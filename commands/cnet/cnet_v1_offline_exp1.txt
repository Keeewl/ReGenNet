==================== Train ====================
1. Train in 4 GPU:
export CUDA_VISIBLE_DEVICES=4,5,6,7
mpiexec -n 4 --allow-run-as-root python -m train.train_mdm \
  --setting cnet \
  --save_dir save/cnet/chi3d_smplx_offline_exp1 \
  --dataset chi3d \
  --cond_mask_prob 0 \
  --num_person 2 \
  --layers 7 \
  --num_frames 150 \
  --arch offline \
  --cm_mode concat \
  --overwrite \
  --pose_rep rot6d \
  --body_model smplx \
  --data_path dataset/chi3d/chi3d_smplx_train.h5 \
  --train_platform_type TensorboardPlatform \
  --vel_threshold 0.01 \
  --unconstrained \
  --resume_checkpoint save/cnet/xxxxx/model000100000.pt

*. Visualization
tensorboard --logdir save/cnet/chi3d_smplx_offline_v1





==================== Motion Synthesis and Visualize ====================
1. Motion Synthesis
python -m sample.cgenerate \
  --model_path save/cnet/chi3d_smplx_offline_exp1//model000120000.pt \
  --dataset chi3d \
  --body_model smplx \
  --pose_rep rot6d \
  --num_person 2 \
  --data_path dataset/chi3d/chi3d_smplx_train.h5 \
  --action_file assets/action_names_chi3d.txt \
