==================== Train ====================
1. Train in 4 GPUs (online):
export CUDA_VISIBLE_DEVICES=4,5,6,7
mpiexec -n 4 --allow-run-as-root python -m train.train_mdm \
  --setting cmdm \
  --save_dir save/cmdm/chi3d_smplx_online_exp1 \
  --dataset chi3d \
  --cond_mask_prob 0 \
  --num_person 2 \
  --layers 8 \
  --num_frames 150 \
  --arch online \
  --overwrite \
  --pose_rep rot6d \
  --body_model smplx \
  --data_path dataset/chi3d/chi3d_smplx_train.h5 \
  --train_platform_type TensorboardPlatform \
  --vel_threshold 0.01 \
  --unconstrained \
  --resume_checkpoint save/cmdm/chi3d_smplx_online_exp1/model000120000.pt

*. Visualization
tensorboard --logdir save/cmdm/chi3d_smplx_online_exp1





==================== Evaluation ====================
export CUDA_VISIBLE_DEVICES=0
python -m eval.eval_cmdm \
  --model_path save/cmdm/chi3d_smplx_online_exp1/model000120000.pt \
  --rec_model_path recognition_training/chi3d_smplx_exp1/checkpoint_0100.pth.tar \
  --eval_mode full \
  --use_ddim \
  --device 0 --batch_size 64 \
  --timestep_respacing ddim5

*. Get a table:
DDPM:
python -m eval.easy_table save/cmdm/chi3d_smplx_online_exp1/evaluation_results_iter120000_samp1000_scale1_a2m_not_auto__full.yaml
DDIM:
python -m eval.easy_table save/cmdm/chi3d_smplx_online_exp1/evaluation_results_iter120000_samp1000_scale1_a2m_not_auto_ddim5_full.yaml
