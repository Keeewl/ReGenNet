==================== Train ====================
1. Train in 4 GPUs:
export CUDA_VISIBLE_DEVICES=4,5,6,7
mpiexec -n 4 --allow-run-as-root python -m train.train_mdm \
  --setting cmdm \
  --save_dir save/cmdm/chi3d_smplx_offline_exp1 \
  --dataset chi3d \
  --cond_mask_prob 0 \
  --num_person 2 \
  --layers 8 \
  --num_frames 150 \
  --arch offline \
  --overwrite \
  --pose_rep rot6d \
  --body_model smplx \
  --data_path dataset/chi3d/chi3d_smplx_train.h5 \
  --train_platform_type TensorboardPlatform \
  --vel_threshold 0.01 \
  --unconstrained \
  --resume_checkpoint save/cmdm/chi3d_smplx_offline_exp1/model000100000.pt

*. Visualization
tensorboard --logdir save/cmdm/chi3d_smplx_offline_exp1





==================== Evaluation ====================
export CUDA_VISIBLE_DEVICES=7
python -m eval.eval_cmdm \
  --model_path save/cmdm/chi3d_smplx_offline_exp1/model000110000.pt \
  --rec_model_path recognition_training/chi3d_smplx_mytrain/checkpoint_0030.pth.tar \
  --eval_mode full \
  --use_ddim \
  --device 0 --batch_size 64
  --timestep_respacing ddim5

*. Get a table:
python -m eval.easy_table save/chi3d/evaluation_results_iter568404_samp1000_scale1_a2m_not_auto__full.yaml
python -m eval.easy_table save/chi3d/evaluation_results_iter568404_samp1000_scale1_a2m_not_auto_ddim5_full.yaml

python -m eval.easy_table save/cmdm/chi3d_smplx_offline_exp1/evaluation_results_iter120000_samp1000_scale1_a2m_not_auto__full.yaml
