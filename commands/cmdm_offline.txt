==================== Train ====================
1. Train in 1 GPU:
python -m train.train_mdm \
--setting cmdm \
--save_dir save/cmdm/chi3d_smplx \
--dataset chi3d \
--cond_mask_prob 0 \
--num_person 2 \
--layers 8 \
--num_frames 150 \
--arch offline \
--overwrite \
--pose_rep rot6d \
--body_model smplx \
--data_path dataset/chi3d/chi3d_smplx_train.h5 \
--train_platform_type TensorboardPlatform \
--vel_threshold 0.01 \
--unconstrained

mpiexec -n 4 --allow-run-as-root python -m train.train_mdm \
