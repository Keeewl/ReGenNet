==================== Train ====================
1. Train in 4 GPU:
export CUDA_VISIBLE_DEVICES=4,5,6,7
mpiexec -n 8 --allow-run-as-root python -m train.train_mdm \
  --setting cnet \
  --save_dir save/cnet/chi3d_smplx_offline_4gpu_v2 \
  --dataset chi3d \
  --cond_mask_prob 0 \
  --num_person 2 \
  --layers 8 \
  --num_frames 150 \
  --arch offline \
  --cm_mode concat \
  --overwrite \
  --pose_rep rot6d \
  --body_model smplx \
  --data_path dataset/chi3d/chi3d_smplx_train.h5 \
  --train_platform_type TensorboardPlatform \
  --vel_threshold 0.01 \
  --unconstrained

*. Visualization
tensorboard --logdir save/cnet/chi3d_smplx_offline_4gpu


==================== Motion Synthesis and Visualize ====================

python -m sample.cgenerate \
